get_contrasts <- function(m) {
  dat_con <- data.frame()
  for(par in c("response", "mu", "confright", "confleft", "precright", "precleft", "pmid", "pex", "bex")) {
    cat(par, " * ")
    dat_con <- rbind(
      dat_con,
      mutate(estimate_contrasts(m, contrast = "Condition", predict = par, backend = "emmeans", test = "pd"), Parameter = par)
      )
  }
  
  rez_con <- dat_con |> 
    filter(pd > .99) |> 
    mutate(Contrast = paste(Level1, "-", str_remove(str_remove(Level2, "\\("), "\\)"))) |> 
    format_table(zap_small = TRUE) |> 
    mutate(param_string = paste0(Parameter, " (", Difference, ", ", CI, ", ", pd, ")")) |> 
    summarise(Parameters = paste(param_string, collapse = "; "), .by = "Contrast")
  
  rez_con
}

get_marginal_parameters <- function(m) {
  # 1. Get predicted averages
  dat_par <- estimate_means(m, by = "Condition", predict = "response", backend = "emmeans", test = NULL) |>
    select(Condition, Mean = Probability, Mean_low = CI_low, Mean_high = CI_high)
  
  # 2. Get predicted parameters
  for (par in c("mu", "confright", "confleft", "precleft", "precright", "pmid", "pex", "bex")) {
    cat(par, " * ")
      dat_par <- estimate_means(m, by = "Condition", predict = par, backend = "emmeans", test = NULL) |>
      datawizard::data_rename(
        select = c("CI_low", "CI_high"),
        replacement = paste0(tools::toTitleCase(par), c("_low", "_high"))
      ) |>
      full_join(dat_par, by = "Condition")
    }
  dat_par
}


get_marginal_densities <- function(dat_par, adjust = 2) {
  
  # 3. Get corresponding densities
  dat_dens <- data.frame()
  dat_annot <- data.frame()
  for (row in 1:nrow(dat_par)) {
    x <- seq(0, 1, length.out = 201)
    # Compute density line
    y <- dchoco(
      x = x, p = dat_par$Mu[row], confleft = dat_par$Confleft[row], confright = dat_par$Confright[row],
      precleft = dat_par$Precleft[row], precright = dat_par$Precright[row],
      pmid = dat_par$Pmid[row], pex = dat_par$Pex[row], bex = dat_par$Bex[row]
    ) 
    y <- y * adjust # Aesthetic adjustment for the plot so that it matches the histogram
    dat_dens <- rbind(dat_dens, data.frame(Condition = dat_par$Condition[row], x = x, y = y))
  
    # Find particular points coordinates
    dat_annot <- rbind(dat_annot, data.frame(
      Condition = dat_par$Condition[row],
      right_y = max(y[x > 0.5]), left_y = max(y[x < 0.5]),
      right_x = 0.5 + x[which.max(y[x > 0.5])], left_x = x[which.max(y[x < 0.5])],
      right_size = dat_par$Mu[row], left_size = 1 - dat_par$Mu[row]
    ))
  }
  
  
  offset <- 1 - (rev(as.numeric(dat_par$Condition)) / 3)
  max_y <- max(dat_dens$y)
  dat_par$Mean_y <- 1.2 * max_y - offset * (1.2 * max_y - max_y)
  
  list(par=dat_par, dens=dat_dens, annot=dat_annot)
}


get_choco_plot <- function(dat_par, observed, adjust = 2, xlabs = c("0% - Ugly", "Beauty", "100% - Beautiful")) {
  dat <- get_marginal_densities(dat_par, adjust = adjust)
    
  dat$dens |>
    ggplot(aes(x = x, y = y)) +
    geom_histogram(data = data.frame(obs = observed[!is.na(observed)]), aes(x = obs, y = after_stat(density)), alpha = 0.1, bins = 47) +
    geom_segment(data = dat$annot, aes(
      x = right_x, xend = right_x,
      y = 0, yend = right_y, color = Condition
    ), linewidth = 0.5, linetype = "dashed") +
    geom_segment(data = dat$annot, aes(
      x = left_x, xend = left_x,
      y = 0, yend = left_y, color = Condition
    ), linewidth = 0.5, linetype = "dashed") +
    geom_point(data = dat$annot, aes(x = right_x, y = 0, color = Condition, size = right_size * 6)) +
    geom_point(data = dat$annot, aes(x = left_x, y = 0, color = Condition, size = left_size * 6)) +
    geom_pointrange(data = dat$par, aes(x = Mean, y = Mean_y, xmin = Mean_low, xmax = Mean_high, color = Condition), linewidth = 0.5) +
    geom_line(aes(color = Condition), linewidth = 1) +
    labs(y = "Likelihood of Response", x = xlabs[2]) +
    scale_size_identity() +
    scale_color_manual(values = cols) +
    scale_x_continuous(breaks = c(0, 0.25, 0.5, 0.75, 1), 
                       labels = c(xlabs[1], "25%", "50%",  "75%", xlabs[3])) +
    theme_minimal() +
    theme(axis.text.y = element_blank())
}

# ─── Frame plan ───────────────────────────────────────────────────────────────
F_LINE  = 55   # frames per density line
PAUSE   = 10   # quiet gap between phases
F_SLIDE = 40   # pointranges slide in
F_DROP  = 40   # vertical lines fall
F_DOT   = 18   # bottom dots grow in
TAIL    = 30   # hold at end

p1_end   = 3 * F_LINE                  # 165

# ─── Easing ───────────────────────────────────────────────────────────────────
def ease_out(t: float) -> float:
    t = max(0.0, min(1.0, t))
    return 1 - (1 - t) ** 2

def ease_in_out(t: float) -> float:
    t = max(0.0, min(1.0, t))
    return t * t * (3 - 2 * t)
  
# ─── Update ───────────────────────────────────────────────────────────────────
def update(frame: int):

    # ── Phase 1: draw density lines one by one, left→right ────────────────────
    for i, cond in enumerate(conditions):
        fs, fe = i * F_LINE, (i + 1) * F_LINE
        d = dens[cond]
        if frame >= fe:
            dens_lines[cond].set_data(d["x"], d["y"])
        elif frame >= fs:
            t = ease_out((frame - fs) / F_LINE)
            n = max(1, int(t * len(d)))
            dens_lines[cond].set_data(d["x"].iloc[:n], d["y"].iloc[:n])

    # ── Phase 2: all three pointranges slide in from the left simultaneously ───
    if frame >= p2_start:
        t = ease_out((frame - p2_start) / F_SLIDE)
        for cond in conditions:
            p = par[cond]
            low, high, mean, y_pr = p["Mean_low"], p["Mean_high"], p["Mean"], p["Mean_y"]
            cur_hi = low + t * (high - low)
            pr_segs[cond].set_data([low, cur_hi], [y_pr, y_pr])
            # Centre dot appears once the sliding front reaches the mean position
            frac = (mean - low) / max(high - low, 1e-9)
            if t >= frac:
                dot_t = ease_out((t - frac) / max(1.0 - frac, 1e-9))
                pr_dots[cond].set_data([mean], [y_pr])
                pr_dots[cond].set_markersize(5.5 * dot_t)
            else:
                pr_dots[cond].set_data([], [])

    # ── Phase 3: vertical lines fall downward from the density peak ────────────
    # Top stays pinned at the curve; bottom drops from peak → 0
    if frame >= p3_start:
        t = ease_out((frame - p3_start) / F_DROP)
        for cond in conditions:
            a = annot[cond]
            bot_r = a["right_y"] * (1 - t)   # bottom falls from right_y to 0
            bot_l = a["left_y"]  * (1 - t)
            vl_right[cond].set_data([a["right_x"], a["right_x"]], [bot_r, a["right_y"]])
            vl_left[cond].set_data( [a["left_x"],  a["left_x"] ], [bot_l, a["left_y"] ])

    # ── Phase 4: bottom dots grow in at y = 0 ─────────────────────────────────
    # Sizes mirror R's geom_point(aes(size = right_size * 6)) with scale_size_identity()
    # ggplot size in mm → matplotlib markersize in pts: × 2.835
    if frame >= p4_start:
        t = ease_out((frame - p4_start) / F_DOT)
        for cond in conditions:
            a = annot[cond]
            ms_r = a["right_size"] * 6 * 2.835 * t
            ms_l = a["left_size"]  * 6 * 2.835 * t
            bdot_right[cond].set_data([a["right_x"]], [0])
            bdot_right[cond].set_markersize(max(0.1, ms_r))
            bdot_left[cond].set_data( [a["left_x"]],  [0])
            bdot_left[cond].set_markersize(max(0.1, ms_l))
            
    # ── Slider handle: middle → wait → drag left → drag right to the mean ─────
    target_x = par["Human Original"]["Mean"]
    if frame <= h_wait_end:
        x = HANDLE_START
    elif frame <= h_left_end:
        t = ease_in_out((frame - h_wait_end) / F_DRAG_LEFT)
        x = HANDLE_START + t * (HANDLE_LEFT - HANDLE_START)
    elif frame <= h_pause_end:
        x = HANDLE_LEFT
    elif frame <= h_end:
        t = ease_in_out((frame - h_pause_end) / F_DRAG_RIGHT)
        x = HANDLE_LEFT + t * (target_x - HANDLE_LEFT)
    else:
        x = target_x

    handle_text.set_x(x)
    pct_text.set_x(x)
    pct_text.set_text(f"{x * 100:.0f}%")

    return (
        list(dens_lines.values())
        + list(pr_segs.values())
        + list(pr_dots.values())
        + list(vl_left.values())
        + list(vl_right.values())
        + list(bdot_left.values())
        + list(bdot_right.values())
        + [handle_text, pct_text]
    )
