get_contrasts <- function(m, outcome = "Beauty", contrast = "Condition", by = NULL, backend = "emmeans") {
  dat_con <- data.frame()
  
  params <- c("response", "mu", insight::find_auxiliary(m))
  if(outcome %in% c("Worth", "SelfRelevance")) params <- params[!params %in% c("disc")]
  if(outcome %in% c("pLeft", "pCenter")) params <- params[!params %in% c("zoi", "coi")]
  for(par in params) {
    if(!is.null(by) && par %in% c("pmid", "pex", "bex", "disc")) next
    # cat(par, " * ")
    
    c <- estimate_contrasts(m, contrast = contrast, by = by, predict = par, 
                            backend = backend, test = "pd", iterations = 500) 
    
    if(outcome %in% c("Worth", "SelfRelevance") && par == "response") {
      c <- filter(c, Response1 == Response2)
      c <- mutate(c, Parameter = paste0("response - ", Response1))
      c <- data_remove(c, c("Response1", "Response2"))
    } else {
      c <- mutate(c, Parameter = par)
    }
    
    dat_con <- rbind(dat_con, c)
  }
  
  if(backend == "marginaleffects") dat_con <- data_rename(dat_con, select = "Median", "Difference")
  dat_con$Outcome <- outcome
  dat_con$Contrast <- paste(str_remove(str_remove(dat_con$Level1, "\\("), "\\)"), 
                            "-", 
                            str_remove(str_remove(dat_con$Level2, "\\("), "\\)")) 
  
  rez_con <- filter(dat_con, sign(CI_low) == sign(CI_high)) 
  if(nrow(rez_con) == 0) {
    rez_con <- data.frame(Contrast = unique(dat_con$Contrast), Parameters = "None")
  } else {
    rez_con <- rez_con |>
      format_table(zap_small = TRUE) |> 
      mutate(param_string = paste0(Parameter, " (", Difference, ", ", CI, ")")) |> 
      summarise(Parameters = paste(param_string, collapse = "; "), .by = "Contrast")
  }
  
  list(rez_con = rez_con, dat_con = dat_con)
}

get_marginal_parameters <- function(m, by = "Condition") {
  # 1. Get predicted averages
  dat_par <- estimate_means(m, by = by, predict = "response", backend = "emmeans", test = NULL) |>
    datawizard::data_rename(
      select = c("Probability", "CI_low", "CI_high"),
      replacement = c("Mean", "Mean_low", "Mean_high")
    ) 
  
  # 2. Get predicted parameters
  for (par in c("mu", "confright", "confleft", "precright", "precleft", "pex", "bex", "pmid")) {
    # cat(par, " * ")
    if(par %in% c("pex", "bex", "pmid")) by <- "Condition" # Overwrite for last params
      dat_par <- estimate_means(m, by = by, predict = par, backend = "emmeans", test = NULL) |>
      datawizard::data_rename(
        select = c("CI_low", "CI_high"),
        replacement = paste0(tools::toTitleCase(par), c("_low", "_high"))
      ) |>
      full_join(dat_par, by = by)
    }
  dat_par
}


get_marginal_densities <- function(dat_par, adjust = 2) {
  
  # 3. Get corresponding densities
  dat_dens <- data.frame()
  dat_annot <- data.frame()
  for (row in 1:nrow(dat_par)) {
    x <- seq(0, 1, length.out = 201)
    # Compute density line
    y <- dchoco(
      x = x, p = dat_par$Mu[row], confleft = dat_par$Confleft[row], confright = dat_par$Confright[row],
      precleft = dat_par$Precleft[row], precright = dat_par$Precright[row],
      pmid = dat_par$Pmid[row], pex = dat_par$Pex[row], bex = dat_par$Bex[row]
    ) 
    y <- y * adjust # Aesthetic adjustment for the plot so that it matches the histogram
    dat_dens <- rbind(dat_dens, data.frame(Condition = dat_par$Condition[row], x = x, y = y))
  
    # Find particular points coordinates
    dat_annot <- rbind(dat_annot, data.frame(
      Condition = dat_par$Condition[row],
      right_y = max(y[x > 0.5]), left_y = max(y[x < 0.5]),
      right_x = 0.5 + x[which.max(y[x > 0.5])], left_x = x[which.max(y[x < 0.5])],
      right_size = dat_par$Mu[row], left_size = 1 - dat_par$Mu[row]
    ))
  }
  
  
  offset <- 1 - (rev(as.numeric(dat_par$Condition)) / 3)
  max_y <- max(dat_dens$y)
  dat_par$Mean_y <- 1.2 * max_y - offset * (1.2 * max_y - max_y)
  
  list(par=dat_par, dens=dat_dens, annot=dat_annot)
}


get_choco_plot <- function(dat, title = "Beauty", observed = title, xlabs = c("0% - Ugly", "100% - Beautiful"), facet = NULL, ...) {
   dat_obs <- dftask
   dat_obs$obs <- dat_obs[[observed]]
   dat_obs <- dat_obs[!is.na(dat_obs$obs), ]
  
  p <- dat$dens |>
    ggplot(aes(x = x, y = y)) +
    geom_histogram(data = dat_obs, aes(x = obs, y = after_stat(density)), alpha = 0.1, bins = 47) +
    geom_segment(data = dat$annot, aes(
      x = right_x, xend = right_x,
      y = 0, yend = right_y, color = Condition
    ), linewidth = 0.5, linetype = "dashed") +
    geom_segment(data = dat$annot, aes(
      x = left_x, xend = left_x,
      y = 0, yend = left_y, color = Condition
    ), linewidth = 0.5, linetype = "dashed") +
    geom_point(data = dat$annot, aes(x = right_x, y = 0, color = Condition, size = right_size * 6)) +
    geom_point(data = dat$annot, aes(x = left_x, y = 0, color = Condition, size = left_size * 6)) +
    geom_pointrange(data = dat$par, aes(x = Mean, y = Mean_y, xmin = Mean_low, xmax = Mean_high, color = Condition), linewidth = 0.5) +
    geom_line(aes(color = Condition), linewidth = 1) +
    labs(y = "Likelihood of Response", title = title) +
    scale_size_identity() +
    scale_color_manual(values = cols) +
    scale_x_continuous(breaks = c(0, 0.25, 0.5, 0.75, 1), 
                       labels = c(xlabs[1], "25%", "50%",  "75%", xlabs[2])) +
    theme_minimal() +
    theme(axis.text.y = element_blank(),
          axis.title.x = element_blank(),
          axis.title.y = element_blank(),
          plot.title = element_text(face = "bold"))
  
  if(!is.null(facet)) p <- p + facet_wrap(facet, ...)
  p
}

get_discrete_plot <- function(pred, title = "Valence", observed = title, facet = NULL, ...) {
   dat_obs <- dftask
   dat_obs$obs <- dat_obs[[observed]]
   dat_obs <- dat_obs[!is.na(dat_obs$obs), ]
   
   if(title == "Worth") dat_obs$obs <- as.numeric(dat_obs$obs)
   
   stratifiers <- c("Condition")
   if(!is.null(facet)) stratifiers <- c(stratifiers, "Emotion")
   
   pred_long <- reshape_iterations(pred) |>
     summarize(N = n(), .by = all_of(c(stratifiers, "iter_group", "iter_value"))) |>
      mutate(
        p = N / sum(N),
        .by = all_of(c(stratifiers, "iter_group"))
      ) |>
      summarise(
        median = median(p),
        lower = quantile(p, .025),
        upper = quantile(p, .975),
        .by = all_of(c(stratifiers, "iter_value"))
      )
  
  p <- dat_obs |>
    ggplot(aes(x = obs)) +
    geom_bar(alpha = 0.5, aes(y = after_stat(prop))) +
    geom_line(
    data = pred_long,
    aes(x = iter_value, y = median, group = Condition, colour = Condition), 
    position = position_dodge(width = 0.25, reverse = TRUE), linewidth = 1
  )  +
    geom_pointrange(
    data = pred_long,
    aes(x = iter_value, y = median,  ymin = lower,  ymax = upper,  color = Condition),
    position = position_dodge(width = 0.25, reverse = TRUE)
  ) +
    labs(y = "Likelihood of Response", title = title) +
    scale_size_identity() +
    scale_color_manual(values = cols) +
    theme_minimal() +
    theme(axis.text.y = element_blank(),
          axis.title.x = element_blank(),
          axis.title.y = element_blank(),
          plot.title = element_text(face = "bold"))
  
  if(!is.null(facet)) p <- p + facet_wrap(facet, ...)
  p
}

# ─── Easing ───────────────────────────────────────────────────────────────────
def ease_out(t: float) -> float:
    t = max(0.0, min(1.0, t))
    return 1 - (1 - t) ** 2

def ease_in_out(t: float) -> float:
    t = max(0.0, min(1.0, t))
    return t * t * (3 - 2 * t)
  
# ─── Update ───────────────────────────────────────────────────────────────────
def update(frame: int):

    # ── Phase 1: draw density lines one by one, left→right ────────────────────
    for i, cond in enumerate(conditions):
        fs, fe = i * F_LINE, (i + 1) * F_LINE
        d = dens[cond]
        if frame >= fe:
            dens_lines[cond].set_data(d["x"], d["y"])
        elif frame >= fs:
            t = ease_out((frame - fs) / F_LINE)
            n = max(1, int(t * len(d)))
            dens_lines[cond].set_data(d["x"].iloc[:n], d["y"].iloc[:n])

    # ── Phase 2: all three pointranges slide in from the left simultaneously ───
    if frame >= p2_start:
        t = ease_out((frame - p2_start) / F_SLIDE)
        for cond in conditions:
            p = par[cond]
            low, high, mean, y_pr = p["Mean_low"], p["Mean_high"], p["Mean"], p["Mean_y"]
            cur_hi = low + t * (high - low)
            pr_segs[cond].set_data([low, cur_hi], [y_pr, y_pr])
            # Centre dot appears once the sliding front reaches the mean position
            frac = (mean - low) / max(high - low, 1e-9)
            if t >= frac:
                dot_t = ease_out((t - frac) / max(1.0 - frac, 1e-9))
                pr_dots[cond].set_data([mean], [y_pr])
                pr_dots[cond].set_markersize(5.5 * dot_t)
            else:
                pr_dots[cond].set_data([], [])

    # ── Phase 3: vertical lines fall downward from the density peak ────────────
    # Top stays pinned at the curve; bottom drops from peak → 0
    if frame >= p3_start:
        t = ease_out((frame - p3_start) / F_DROP)
        for cond in conditions:
            a = annot[cond]
            bot_r = a["right_y"] * (1 - t)   # bottom falls from right_y to 0
            bot_l = a["left_y"]  * (1 - t)
            vl_right[cond].set_data([a["right_x"], a["right_x"]], [bot_r, a["right_y"]])
            vl_left[cond].set_data( [a["left_x"],  a["left_x"] ], [bot_l, a["left_y"] ])

    # ggplot size in mm → matplotlib markersize in pts: × 2.835
    if frame >= p4_start:
        t = ease_out((frame - p4_start) / F_DOT)
        for cond in conditions:
            a = annot[cond]
            ms_r = a["right_size"] * 6 * 2.835 * t
            ms_l = a["left_size"]  * 6 * 2.835 * t
            bdot_right[cond].set_data([a["right_x"]], [0])
            bdot_right[cond].set_markersize(max(0.1, ms_r))
            bdot_left[cond].set_data( [a["left_x"]],  [0])
            bdot_left[cond].set_markersize(max(0.1, ms_l))
            
    # ── Slider handle: middle → wait → drag left → drag right to the mean ─────

    elif frame <= h_left_end:

def star_vertices(cx, cy, outer_r, spikiness, n_points=5, rotation=np.pi / 2):
    spikiness = np.clip(spikiness, 0.0, 1.0)
    inner_r = outer_r * (1 - spikiness * 0.62)
    n = n_points * 2
    angles = rotation + np.arange(n) * (2 * np.pi / n)
    radii = np.where(np.arange(n) % 2 == 0, outer_r, inner_r)
    return np.column_stack([cx + radii * np.cos(angles), cy + radii * np.sin(angles)])

# ─── Easing / interpolation helpers ───────────────────────────────────────────
def ease_in_out(t: float) -> float:
    t = max(0.0, min(1.0, t))
    return t * t * (3 - 2 * t)

def lerp(a, b, t):
    return a + t * (b - a)

def lerp_color(c0, c1, t):
    rgb0, rgb1 = np.array(to_rgb(c0)), np.array(to_rgb(c1))
    return tuple(lerp(rgb0, rgb1, t))

def update(frame: int):

    local = frame % SEG
    cur_state = state_order[s]
    nxt_state = state_order[(s + 1) % N_STATES]

def render_frame_image():
    fig.canvas.draw()
    buf = np.asarray(fig.canvas.buffer_rgba())
    return Image.fromarray(buf).convert("RGB")

# palette actually contains black, green, red, and the in-between hues.

    [0, N_FRAMES - 1] +
    [i * SEG for i in range(N_STATES)] +
    [i * SEG + F_HOLD // 2 for i in range(N_STATES)] +
    [i * SEG + F_HOLD + F_TRANS // 2 for i in range(N_STATES)]
))
calib_imgs = []
for f in calibration_frames:
    update(f)
    calib_imgs.append(render_frame_image())

calib_strip = Image.new("RGB", (calib_imgs[0].width, calib_imgs[0].height * len(calib_imgs)))
for i, im_ in enumerate(calib_imgs):
    calib_strip.paste(im_, (0, i * im_.height))
